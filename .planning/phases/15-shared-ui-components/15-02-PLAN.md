<task id="1">
Create `app/src/components/shared/UploadZone.tsx`

A drag-and-drop upload zone with dashed border, preview, validation, and file format hints.

```tsx
import React, { useRef, useState, useCallback } from 'react';
import { UploadCloud, X } from 'lucide-react';

interface UploadZoneProps {
  onFileSelect: (file: { data: string; name: string; type: string }) => void;
  accept?: string;
  maxSizeMB?: number;
  preview?: { data: string; name: string } | null;
  onClear?: () => void;
  className?: string;
  compact?: boolean;
}

export function UploadZone({
  onFileSelect,
  accept = 'image/*',
  maxSizeMB = 10,
  preview,
  onClear,
  className = '',
  compact = false,
}: UploadZoneProps) {
  const fileInputRef = useRef<HTMLInputElement>(null);
  const [isDragging, setIsDragging] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const processFile = useCallback((file: File) => {
    setError(null);
    if (maxSizeMB && file.size > maxSizeMB * 1024 * 1024) {
      setError(`File too large. Max ${maxSizeMB}MB.`);
      return;
    }
    const reader = new FileReader();
    reader.onload = (e) => {
      onFileSelect({
        data: e.target?.result as string,
        name: file.name,
        type: file.type,
      });
    };
    reader.readAsDataURL(file);
  }, [onFileSelect, maxSizeMB]);

  const handleDrop = useCallback((e: React.DragEvent) => {
    e.preventDefault();
    setIsDragging(false);
    const file = e.dataTransfer.files?.[0];
    if (file) processFile(file);
  }, [processFile]);

  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (file) processFile(file);
  };

  if (preview) {
    return (
      <div className={`relative bg-bg-surface2 rounded-[16px] border border-border-default overflow-hidden ${className}`}>
        <img src={preview.data} alt={preview.name} className="w-full max-h-[400px] object-contain p-2 rounded-[16px]" />
        {onClear && (
          <button
            onClick={onClear}
            className="absolute top-3 right-3 bg-black/50 text-white p-1.5 rounded-full hover:bg-black/70 transition-colors"
            aria-label="Remove image"
          >
            <X className="w-4 h-4" />
          </button>
        )}
        <p className="px-4 pb-3 text-[12px] text-text-tertiary truncate">{preview.name}</p>
      </div>
    );
  }

  return (
    <div className={className}>
      <div
        className={`border-2 border-dashed rounded-[16px] flex flex-col items-center justify-center text-center cursor-pointer transition-colors ${
          compact ? 'py-6 px-4' : 'py-12 px-6'
        } ${
          isDragging
            ? 'border-accent bg-accent-subtle'
            : 'border-border-default hover:border-accent bg-bg-surface1'
        }`}
        onClick={() => fileInputRef.current?.click()}
        onDragOver={(e) => { e.preventDefault(); setIsDragging(true); }}
        onDragLeave={() => setIsDragging(false)}
        onDrop={handleDrop}
      >
        <UploadCloud className={`text-text-tertiary mb-3 ${compact ? 'w-8 h-8' : 'w-12 h-12'}`} />
        <p className={`font-medium text-text-primary mb-1 ${compact ? 'text-[14px]' : 'text-[16px]'}`}>
          Drop your design here
        </p>
        <p className="text-[14px] text-text-secondary mb-1">or click to browse</p>
        <p className="text-[12px] text-text-tertiary">PNG, JPG, WEBP · Max {maxSizeMB}MB</p>
        <input
          type="file"
          ref={fileInputRef}
          className="hidden"
          accept={accept}
          onChange={handleFileChange}
        />
      </div>
      {error && (
        <p className="mt-2 text-[12px] text-error">{error}</p>
      )}
    </div>
  );
}
```

**Key decisions:**
- Two visual modes: empty zone (dashed border) and preview (image + clear button)
- `compact` prop for smaller contexts (e.g., compare slots)
- Built-in client-side file size validation
- Krea-style drag state: accent border + subtle accent background
- Preview state: image with X button overlay (only on hover would need CSS, using always-visible for clarity)
</task>

Create `app/src/components/shared/LoadingSpinner.tsx`

Two loading patterns: a spinning indicator and a skeleton card placeholder.

```tsx
import React from 'react';

interface SpinnerProps {
  size?: 'sm' | 'md' | 'lg';
  className?: string;
}

export function Spinner({ size = 'md', className = '' }: SpinnerProps) {
  const sizeStyles = {
    sm: 'w-4 h-4 border-2',
    md: 'w-6 h-6 border-2',
    lg: 'w-8 h-8 border-[3px]',
  };

  return (
    <div
      className={`rounded-full border-accent border-t-transparent animate-spin ${sizeStyles[size]} ${className}`}
      role="status"
      aria-label="Loading"
    />
  );
}

interface SkeletonProps {
  className?: string;
}

export function Skeleton({ className = '' }: SkeletonProps) {
  return (
    <div className={`bg-bg-surface2 rounded-[8px] animate-pulse ${className}`} />
  );
}

interface SkeletonCardProps {
  lines?: number;
  className?: string;
}

export function SkeletonCard({ lines = 3, className = '' }: SkeletonCardProps) {
  return (
    <div className={`bg-bg-surface1 border border-border-default rounded-[16px] p-6 flex flex-col gap-3 ${className}`}>
      <Skeleton className="h-5 w-[60%]" />
      {Array.from({ length: lines }).map((_, i) => (
        <Skeleton
          key={i}
          className={`h-4 ${i === lines - 1 ? 'w-[40%]' : i === lines - 2 ? 'w-[80%]' : 'w-full'}`}
        />
      ))}
    </div>
  );
}
```

**Key decisions:**
- 3 exports: `Spinner` (circular), `Skeleton` (single block), `SkeletonCard` (card with text lines)
- `Spinner` uses accent color border (consistent with current pattern)
- `Skeleton` uses `animate-pulse` (subtle opacity pulse — replaces custom shimmer for simple cases)
- `SkeletonCard` replicates existing skeleton layout patterns from analyzing screens
- `role="status"` + `aria-label` on Spinner for accessibility
</task>
