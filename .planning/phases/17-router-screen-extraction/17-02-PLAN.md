  ```tsx
  import { Routes, Route } from 'react-router-dom';
  import { AnimatePresence } from 'motion/react';
  import { useAuth } from './contexts/AuthContext';
  import LoginScreen from './LoginScreen';
  import RegisterScreen from './RegisterScreen';
  import SettingsModal from './components/SettingsModal';
  import { AppShell } from './components/layout';
  import {
    HomeScreen,
    AnalyzeScreen,
    CompareScreen,
    StudioScreen,
    LibraryScreen,
    ViewAnalysisScreen,
  } from './screens';
  import PatternDetailScreen from './screens/PatternDetailScreen';

  export default function App() {
    const { user, loading, signOut } = useAuth();
    const [authScreen, setAuthScreen] = useState<'login' | 'register'>('login');
    const [isSettingsOpen, setIsSettingsOpen] = useState(false);

    // Listen for open-settings custom event
    useEffect(() => {
      const handler = () => setIsSettingsOpen(true);
      document.addEventListener('open-settings', handler);
      return () => document.removeEventListener('open-settings', handler);
    }, []);

    if (loading) return <LoadingSpinner />;

    if (!user) return <AuthScreens />;

    return (
      <AppShell onSettingsClick={() => setIsSettingsOpen(true)}>
        <Routes>
          <Route path="/" element={<HomeScreen />} />
          <Route path="/analyze" element={<AnalyzeScreen />} />
          <Route path="/studio" element={<StudioScreen />} />
          <Route path="/compare" element={<CompareScreen />} />
          <Route path="/library" element={<LibraryScreen />} />
          <Route path="/library/pattern/:id" element={<PatternDetailScreen />} />
          <Route path="/library/analysis/:id" element={<ViewAnalysisScreen />} />
        </Routes>

        <AnimatePresence>
          {isSettingsOpen && <SettingsModal onClose={() => setIsSettingsOpen(false)} />}
        </AnimatePresence>
      </AppShell>
    );
  }
  ```

  **Key changes:**
  - Remove ALL inline function definitions (UploadScreen, AnalyzingScreen, ResultsScreen, ViewAnalysisScreen)
  - Remove ALL state that was moved to screens: uploadedImage, analysisResult, error, compareImages, compareResult, initialGenMode, selectedPatternId, savedAnalysisId, viewingAnalysisId, currentScreen
  - Remove ALL handlers that moved to screens: handleAnalyze, handleCompareAnalyze, handleCancelAnalysis
  - Remove migration banner from App.tsx — move to HomeScreen or keep as separate component if still needed
  - Remove `useMigratePatterns` hook — move to HomeScreen
  - Keep: auth gate (loading, !user), isSettingsOpen, open-settings listener
  - Keep: `Screen` type export IF still used by other files, otherwise remove
  - Remove unused imports (ArrowLeft, UploadCloud, AlertTriangle, etc.)

  **Migration banner handling:**
